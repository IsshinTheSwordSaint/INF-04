using System;
class KontoBankowe
{
    public double saldo { get; private set; }

    public KontoBankowe()
    {
        this.saldo = 0.0;
    }
    public KontoBankowe(double saldoPoczatkowe)
    {
        this.saldo = saldoPoczatkowe;
    }
    public void Wplac(double kwota)
    {
        if (kwota > 0)
        {
            this.saldo += kwota;
            Console.WriteLine($"Pomylsnie wplacono kwote: {kwota}. Obecny stan konta wynosi: {this.saldo}");
        }
        else
        {
            Console.WriteLine("Podana kwota jest niepoprawna");
        }
    }
    public void Wyplac(double kwota)
    {
        if (kwota < this.saldo && kwota > 0)
        {
            this.saldo -= kwota;
            Console.WriteLine($"Pomyslnie wyplacono kwote: {kwota}. Obecny stan konta wynosi: {this.saldo}");
        }
        else
        {
            Console.WriteLine("Podana kwota jest niepoprawna");
        }
    }
    static void Main()
    {
        KontoBankowe konto = new KontoBankowe();
        konto.Wplac(67);
        konto.Wyplac(9);
        konto.Wplac(-20);
        konto.Wyplac(50);
    }
}
